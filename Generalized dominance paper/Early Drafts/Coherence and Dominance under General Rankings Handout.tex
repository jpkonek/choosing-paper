\section{Setting}

\begin{stmt}{Definitions 1.1--1.4}
\begin{itemize}[leftmargin=*]
\item Worlds $\Omega\ne\varnothing$; models $\M\ne\varnothing$. A \emph{ranking} $L$: a nonempty $\M_L\subseteq\M$ (the \emph{evaluable} models) and a binary relation $\preceq_\omega$ on $\M_L$ for each $\omega\in\Omega$; $m\preceq_\omega m'$: $m$ performs at least as well as $m'$ at $\omega$. $m$, $m'$ are \emph{tied at $\omega$}: $m\preceq_\omega m'$ and $m'\preceq_\omega m$. $L$ is \emph{transitive} if every $\preceq_\omega$ is. For a fixed $L$, ``model'' means a member of $\M_L$.
\item $c$ \emph{weakly dominates} $m$: $c\preceq_\omega m$ for every $\omega$. $c$ \emph{dominates} $m$: $c$ weakly dominates $m$ and they are not tied at every world. $\U_L$: the undominated models.
\item $\Delta_F$: pmfs $p=(p_\omega)_{\omega\in F}$ on a finite nonempty $F\subseteq\Omega$; $\supp p=\{\omega:p_\omega>0\}$; $\delta_\omega$ the point mass at $\omega$; $\Delta\coloneqq\Delta_\Omega$ when $\Omega$ is finite.
\item Coherent models $\C\subseteq\M$, fixed in advance. $\C_L\coloneqq\C\cap\U_L$.
\end{itemize}
\end{stmt}




\subsection*{Axioms}
\begin{stmt}{Definitions 1.5--1.6}
\begin{itemize}[leftmargin=*]
\item A set $\mathcal A$ of models is \emph{closed under ties}: every model tied at every world with a member of $\mathcal A$ is in $\mathcal A$. To say that $L$ is \emph{closed under ties} just means that $\C_L$ is closed under ties.
\item For a family $\Lf$ of rankings. \emph{Evaluability}: every model in $\M$ is evaluable by some $L\in\Lf$. \emph{Coherent admissibility}: every coherent model is in $\C_L$ for some $L\in\Lf$.
\end{itemize}
\end{stmt}

\begin{stmt}{Lemma 1.7}
Let $a$ weakly dominate $m$. (a) If $m$ is undominated, then $m$ and $a$ are tied at every world. (b) If $L$ is closed under ties, $a\in\C_L$ and $m\notin\C_L$, then $a$ dominates $m$. %(c) More generally, if $\mathcal A$ is closed under ties, $a\in\mathcal A$ and $m\notin\mathcal A$, then $a$ dominates $m$.
\end{stmt}

\end{stmt}

\textbf{Proof of Lemma 2.1.}
\begin{steps}
\item $\delta_\omega\in A_\omega$, since $\supp\delta_\omega=\{\omega\}$. So each $A_\omega\ne\varnothing$.
\item $d_\omega(p)\coloneqq\min_{q\in A_\omega}\lVert p-q\rVert$, $s(p)\coloneqq\sum_{\omega\in F}d_\omega(p)$, $G(p)_\omega\coloneqq(p_\omega+d_\omega(p))/(1+s(p))$. $G:\Delta_F\to\Delta_F$ is continuous.
\item Brouwer: $G(p)=p$ for some $p$; equivalently $p_\omega\, s(p)=d_\omega(p)$ for all $\omega\in F$.
\item Choose $\omega\in\supp p$ with $p\in A_\omega$, where $p$ is our fixed point. Then $d_\omega(p)=0$ and $p_\omega>0$. So $s(p)=0$. Hence $p\in\bigcap_{\omega\in F}A_\omega$.
\end{steps}

\subsection*{Finitely many worlds}
\begin{stmt}{Theorem 2.2}
Let $\Omega$ be finite and $L$ any ranking. Suppose that for each $p\in\Delta$ there is an assigned model $c_p$, not necessarily coherent or undominated, and for each model $m$ and world $\omega$ there is a closed set $A_\omega(m)\subseteq\Delta$, such that:
\begin{enumerate}[label=(\roman*)]
\item $c_p\preceq_\omega m$ for every $p\in A_\omega(m)$;
\item for every model $m$ and every $p\in\Delta$, there is some $\omega\in\supp p$ such that $p\in A_\omega(m)$.
\end{enumerate}
Then for each model $m$, some pmf lies in every $A_\omega(m)$, and the model assigned to any such pmf weakly dominates $m$.
\end{stmt}

\begin{stmt}{Corollary 2.3}
Let $\Omega$ be finite. Suppose that for each $p\in\Delta$ there is an assigned model $c_p$, not necessarily coherent or undominated, such that every model $m$ has some $\omega\in\supp p$ with $c_p\preceq_\omega m$. Suppose further that for any model $m$ and world $\omega$, the set $\{p:c_p\preceq_\omega m\}$ is closed. Then every model is weakly dominated by some $c_p$.
\end{stmt}
\begin{steps}
\item $A_\omega(m)\coloneqq\{p:c_p\preceq_\omega m\}$. In Figure~\ref{fig:finite}: $A_1(m)$ is the set of all $p$ whose $c_p$ lies left of the vertical line, $A_2(m)$ the set of all $p$ whose $c_p$ lies below the horizontal line.
\item Condition (i) of Theorem 2.2 holds by definition of $A_\omega(m)$; condition (ii) by the first assumption. Theorem 2.2 gives the conclusion.
\end{steps}

\subsection*{Arbitrary sets of worlds}
\begin{stmt}{Theorem 2.4}
Let $\Omega$ be any nonempty set and $L$ any ranking. Suppose:
\begin{enumerate}[label=\arabic*.,leftmargin=*]
\item There is a compact space of parameters $\Theta$. For each $\theta\in\Theta$, there is a selected model $c_\theta$, not necessarily coherent or undominated. For each world $\omega$ and model $m$ there is a closed set $K_\omega(m)\subseteq\Theta$ such that $c_\theta\preceq_\omega m$ for every $\theta\in K_\omega(m)$.
\item For every finite set of worlds $G\subseteq\Omega$, there are a nonempty finite set $F\supseteq G$, a map $\theta_F:\Delta_F\to\Theta$, and, for each $\omega\in F$ and model $m$, a closed set $A^F_\omega(m)\subseteq\Delta_F$, such that
\begin{enumerate}[label=(\alph*)]
\item for every $\omega\in F$ and model $m$, $\theta_F(p)\in K_\omega(m)$ for all $p\in A^F_\omega(m)$;
\item for every model $m$ and every $p\in\Delta_F$, there is some $\omega\in\supp p$ such that $p\in A^F_\omega(m)$.
\end{enumerate}
\end{enumerate}
Then every model is weakly dominated by some $c_\theta$.
\end{stmt}

\textbf{Proof.} Fix a model $m$.
\begin{steps}
\item For any $F$ from assumption 2, the sets $A^F_\omega(m)$, $\omega\in F$, satisfy the hypothesis of Lemma 2.1, by 2(b). So some $p_F\in\bigcap_{\omega\in F}A^F_\omega(m)$.
\item By 2(a), $\theta_F(p_F)\in\bigcap_{\omega\in F}K_\omega(m)$.
\item Given $\omega_1,\dots,\omega_k$, apply assumption 2 to $G=\{\omega_1,\dots,\omega_k\}$: $\bigcap_iK_{\omega_i}(m)\supseteq\bigcap_{\omega\in F}K_\omega(m)\ni\theta_F(p_F)$. Every finite subfamily of the closed sets $K_\omega(m)$ has a nonempty intersection.
\item If $\bigcap_{\omega\in\Omega}K_\omega(m)=\varnothing$, the open sets $\Theta\setminus K_\omega(m)$ cover $\Theta$. By compactness, finitely many, $\Theta\setminus K_{\omega_1}(m),\dots,\Theta\setminus K_{\omega_k}(m)$, cover $\Theta$, so $\bigcap_iK_{\omega_i}(m)=\varnothing$, contradicting step 3. So some $\theta^*\in\bigcap_{\omega\in\Omega}K_\omega(m)$.
\item By assumption 1, $c_{\theta^*}\preceq_\omega m$ for every $\omega$: $c_{\theta^*}$ weakly dominates $m$.
\end{steps}

%None of these proofs uses coherence or undominatedness of the assigned models. Weak domination is not yet dominance: an assigned model may be tied at every world with the model it weakly dominates. Corollary 2.5 states what excludes such ties.

\begin{stmt}{Corollary 2.5}
Assume the hypotheses of Theorem 2.2, Corollary 2.3 or Theorem 2.4, and let $\mathcal A$ be the set of assigned models.
\begin{enumerate}[label=(\alph*)]
\item If $\mathcal A$ is closed under ties, every model not in $\mathcal A$ is dominated by a member of $\mathcal A$.
\item If $\mathcal A\subseteq\C_L$, every undominated model is tied at every world with a member of $\C_L$.
\item If $\mathcal A\subseteq\C_L$ and $L$ is closed under ties, then $\U_L=\C_L$, and every other model, in particular every incoherent model, is dominated by a member of $\C_L$.
\end{enumerate}
\end{stmt}

\section{The dominance argument}

\begin{stmt}{Definition 3.1 (Dominance principle)}
Let $\Lf$ be a family of rankings. A model is \emph{irrational} if, under every ranking in $\Lf$ that evaluates it, some model dominates it. A model is \emph{rationally permissible} if it is undominated under some ranking in $\Lf$ that evaluates it.
\end{stmt}

\begin{stmt}{Lemma 3.2}
A model is irrational iff it is not rationally permissible. Under evaluability, an irrational model is dominated under at least one ranking in $\Lf$.
\end{stmt}
\begin{steps}
\item $m$ is not irrational iff some ranking in $\Lf$ that evaluates $m$ leaves $m$ undominated, iff $m$ is rationally permissible.
\item Evaluability: some ranking in $\Lf$ evaluates $m$. If $m$ is irrational, $m$ is dominated under it.
\end{steps}

\begin{stmt}{Definition 3.3 (Assumptions on the rankings)}
The dominance argument assumes the following of the family $\Lf$.
\begin{itemize}[leftmargin=2.4em,labelsep=0.5em]
\item[\textbf{(E)}] \emph{Evaluability}: every model in $\M$ is evaluable by some $L\in\Lf$. No model escapes assessment.
\item[\textbf{(A)}] \emph{Coherent admissibility}: every coherent model is in $\C_L$ for some $L\in\Lf$. Each coherent model is undominated under at least one ranking that evaluates it.
\item[\textbf{(S)}] \emph{Selection}: for each $L\in\Lf$, the hypotheses of Theorem 2.2, Corollary 2.3 or Theorem 2.4 hold, with every assigned model $c_p$, or selected model $c_\theta$, in $\C_L$. Each ranking has enough undominated coherent models to meet the conditions of one of the theorems of Section 2.
\item[\textbf{(T)}] \emph{Closure under ties}: every $L\in\Lf$ is closed under ties, that is, $\C_L$ is. A model tied at every world with an undominated coherent model is itself an undominated coherent model.
\end{itemize}
\end{stmt}

(T) cannot be dropped: with one world, coherent $a$, incoherent $m$ and every comparison holding, (E), (A) and (S) hold for $\Lf=\{L\}$, but $m$ is undominated, so rationally permissible.

\begin{stmt}{Theorem 3.4 (The dominance argument)}
Let $\Lf$ satisfy (E), (A), (S) and (T). Then the rationally permissible models are exactly the coherent ones, and the irrational models are exactly the incoherent ones. Moreover, for each $L\in\Lf$, every incoherent model in $\M_L$ is dominated by a member of $\C_L$, and $\U_L=\C_L$. Every incoherent model is dominated under at least one ranking in $\Lf$.
\end{stmt}

\textbf{Proof of 3.5.} Lemma 1.7 applied to each model and a member of $\C_L$ weakly dominating it. Part (a): every undominated model is tied with a member of $\C_L$; under closure under ties it is then in $\C_L$, so $\U_L=\C_L$. Part (b) gives the last claim.

\textbf{Proof of 3.6.} By Proposition 3.5, every undominated model is tied with some $c\in\C_L$. Conversely:
\begin{steps}
\item Let $m$ be tied with $c\in\C_L$ and $m'$ dominate $m$. Then $m'\preceq_\omega m\preceq_\omega c$, so $m'$ weakly dominates $c$.
\item If $c$ weakly dominated $m'$, then $m\preceq_\omega c\preceq_\omega m'$: $m$ weakly dominates $m'$, contradicting that $m'$ dominates $m$.
\item So $m'$ dominates $c\in\C_L$: impossible. So $m\in\U_L$. A dominated $m$ is not tied with its weak dominator $c\in\C_L$, so $c$ dominates $m$.
\end{steps}

\begin{stmt}{Proposition 3.7}
Suppose that under each $L\in\Lf$, every model in $\M_L$ is weakly dominated by a member of $\C_L$. Then for each $L\in\Lf$: (a) every undominated model is tied at every world with a member of $\C_L$; (b) if $L$ is closed under ties, then $\U_L=\C_L$, and every other model in $\M_L$ is dominated by a member of $\C_L$; (c) if $L$ is transitive, then $\U_L$ consists of the models tied at every world with a member of $\C_L$, and every other model in $\M_L$ is dominated by a member of $\C_L$.

If every ranking in $\Lf$ is closed under ties or transitive, then a model is irrational iff, under every ranking in $\Lf$ that evaluates it, some member of $\C_L$ dominates it. If every ranking in $\Lf$ is closed under ties and coherent admissibility holds, the rationally permissible models are exactly the coherent ones.
\end{stmt}

\textbf{Proof of 3.7.}
\begin{steps}
\item (a), (b): Proposition 3.5. (c): Proposition 3.6.
\item $m$ irrational and $L$ evaluates $m$: $m\notin\U_L$, so by (b) or (c) some member of $\C_L$ dominates $m$. Converse: from the definitions.
\item All $L$ closed under ties: by (b), $\U_L=\C_L\subseteq\C$ for every $L$, so every rationally permissible model is coherent.
\item Coherent admissibility: each coherent model is in some $\C_L\subseteq\U_L\subseteq\M_L$, so it is rationally permissible.
\end{steps}
The last claim is the main conclusion of Theorem 3.4, with (S) replaced by the hypothesis of this subsection; it does not need (E). Proposition 3.7 adds the case of transitive rankings that are not closed under ties, in which irrational models still have dominators in $\C_L$.
